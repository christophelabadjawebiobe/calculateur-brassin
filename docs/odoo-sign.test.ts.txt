type Mode = { sendOk: boolean; hasActionSent: boolean; hasSendAccess: boolean; items: number[]; failSendMsg?: string };
const mode: Mode = { sendOk: true, hasActionSent: true, hasSendAccess: false, items: [1, 2] };
const DB: any = { partners: [{ id: 10, name: 'Odile Existante', email: 'odile_ex@x.be' }, { id: 11, name: 'Sans Mail', email: false }], requests: {} as any, nextP: 100, nextR: 500, log: [] as any[] };
const DOES = (m: string, meth: string) => ({ jsonrpc: '2.0', id: 1, error: { code: 200, message: 'Odoo Server Error', data: { name: 'builtins.AttributeError', message: `The method '${m}.${meth}' does not exist` } } });
const ERR = (name: string, message: string) => ({ jsonrpc: '2.0', id: 1, error: { code: 200, message: 'Odoo Server Error', data: { name, message, debug: 'Traceback...' } } });
function exec(model: string, method: string, args: any[], kw: any): any {
  DB.log.push([model, method, JSON.stringify(args), JSON.stringify(kw)]);
  if (method.startsWith('_')) return ERR('odoo.exceptions.AccessError', 'Private methods cannot be called remotely');
  switch (model + '.' + method) {
    case 'res.partner.search': {
      const dom = args[0]; const em = dom.find((x: any) => x[0] === 'email' && x[1] === '=ilike'); const nm = dom.find((x: any) => x[0] === 'name'); const noem = dom.find((x: any) => x[0] === 'email' && x[1] === '=');
      const unesc = (s: string) => s.replace(/\\(.)/g, '$1').toLowerCase();
      if (em) return { result: DB.partners.filter((p: any) => p.email && p.email.toLowerCase() === unesc(em[2])).map((p: any) => p.id) };
      if (nm && noem) return { result: DB.partners.filter((p: any) => !p.email && p.name.toLowerCase() === unesc(nm[2])).map((p: any) => p.id) };
      return { result: [] };
    }
    case 'res.partner.create': { const id = DB.nextP++; DB.partners.push({ id, ...args[0] }); return { result: id }; }
    case 'res.partner.write': { Object.assign(DB.partners.find((p: any) => p.id === args[0][0]), args[1]); return { result: true }; }
    case 'sign.template.search': return { result: [5] };
    case 'sign.template.read': return { result: args[0][0] === 5 ? [{ id: 5, name: 'Bulletin de souscription', sign_item_ids: mode.items }] : [] };
    case 'sign.item.read': return { result: args[0].map((i: number) => ({ id: i, responsible_id: [3, 'Customer'] })) };
    case 'sign.item.role.read': return { result: args[0].map((i: number) => ({ id: i, name: 'Customer' })) };
    case 'sign.request.fields_get': return { result: { template_id: {}, reference: {}, request_item_ids: {}, subject: {}, message: {}, state: {} } };
    case 'sign.request.item.fields_get': return { result: { partner_id: {}, role_id: {} } };
    case 'sign.request.initialize_new': return DOES('sign.request', 'initialize_new');
    case 'sign.request.create': {
      const v = args[0]; if (!v.template_id || !Array.isArray(v.request_item_ids) || !v.request_item_ids.every((t: any) => t[0] === 0 && t[1] === 0 && t[2].partner_id && t[2].role_id)) return ERR('odoo.exceptions.ValidationError', 'A sign request needs at least one valid sign request item.');
      const id = DB.nextR++; DB.requests[id] = { id, state: 'draft', ...v }; return { result: id };
    }
    case 'sign.request.read': { const r = DB.requests[args[0][0]]; return r ? { result: [{ id: r.id, state: r.state }] } : ERR('odoo.exceptions.MissingError', 'Record does not exist or has been deleted.'); }
    case 'sign.request.action_sent': {
      if (!mode.hasActionSent) return DOES('sign.request', 'action_sent');
      if (!args[0].length) return { result: true };
      if (!mode.sendOk) return ERR('odoo.exceptions.ValidationError', mode.failSendMsg || 'Mail server down');
      DB.requests[args[0][0]].state = 'sent'; return { result: true };
    }
    case 'sign.request.send_signature_accesses': {
      if (!mode.hasSendAccess) return DOES('sign.request', 'send_signature_accesses');
      if (args[0].length) DB.requests[args[0][0]].state = 'sent'; return { result: true };
    }
  }
  return DOES(model, method);
}
const srv = http.createServer((req, res) => { let b = ''; req.on('data', (c) => (b += c)); req.on('end', () => {
  const { params } = JSON.parse(b); let out: any;
  if (params.service === 'common' && params.method === 'authenticate') out = { result: 7 };
  else if (params.service === 'common' && params.method === 'version') out = { result: { server_version: '18.0' } };
  else { const [, , , model, method, a, kw] = params.args; out = exec(model, method, a, kw); }
  res.setHeader('content-type', 'application/json'); res.end(JSON.stringify({ jsonrpc: '2.0', id: 1, ...out }));
}); });
